/**
 * Snap an ISO date to the nearest workable day, looking ±90 days symmetrically.
 * "Workable" means: weekend-skipped iff !includeWeekends, AND not in noWorkSet.
 * Returns the original date if no workable day is found in range (defensive — never throws).
 */
export function snapToNearestWorkable(
  iso: string,
  noWorkSet: Set<string>,
  includeWeekends: boolean,
): string {
  const isWorkable = (candidate: string): boolean => {
    const d = parse(candidate);
    if (!includeWeekends && !isMonFri(d)) return false;
    if (noWorkSet.has(candidate)) return false;
    return true;
  };
  if (isWorkable(iso)) return iso;
  for (let delta = 1; delta <= 90; delta++) {
    const forward = addCalendarDays(iso, delta);
    const backward = addCalendarDays(iso, -delta);
    const fwdOk = isWorkable(forward);
    const bwdOk = isWorkable(backward);
    if (fwdOk && bwdOk) return forward; // ties go forward (matches user "next workday" intuition)
    if (fwdOk) return forward;
    if (bwdOk) return backward;
  }
  return iso;
}
```

import type { ViewportDay } from '../calendar';

/**
 * Pure pixel ↔ date conversions driven by the already-rendered ViewportDay[].
 * The viewport list is the source of truth for which dates have columns; this
 * module never makes calendar decisions of its own.
 */

/** Clamped floor(px/cellW). Returns an index into `days` (0..days.length-1). */
export function pxToDayIndex(px: number, cellW: number, dayCount: number): number {
  if (dayCount === 0) return 0;
  if (cellW <= 0) return 0;
  const raw = Math.floor(px / cellW);
  if (raw < 0) return 0;
  if (raw >= dayCount) return dayCount - 1;
  return raw;
}

/** ISO date of the column under `px`. */
export function pxToDate(px: number, cellW: number, days: ViewportDay[]): string {
  if (days.length === 0) return '';
  const idx = pxToDayIndex(px, cellW, days.length);
  return days[idx].date;
}

/**
 * Left-edge pixel X of the column rendering `iso`. Returns -1 if the date is
 * not currently in the viewport (e.g. weekend with weekends hidden).
 */
export function dateToPx(iso: string, cellW: number, days: ViewportDay[]): number {
  const idx = days.findIndex(d => d.date === iso);
  if (idx === -1) return -1;
  return idx * cellW;
}
```

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { store } from '../store.svelte';
  import { addWorkdays, snapToNearestWorkable } from '../calendar';
  import { pxToDate, dateToPx } from './timeline';
  import * as ipc from '../ipc';
  import type { ViewportDay } from '../calendar';

  let {
    cellW,
    days,
    timelineLeft,
    noWorkSet,
    includeWeekends,
    holidaysBlockWork,
  }: {
    cellW: number;
    days: ViewportDay[];
    timelineLeft: number;
    noWorkSet: Set<string>;
    includeWeekends: boolean;
    holidaysBlockWork: boolean;
  } = $props();

  /** Effective "skip this date" set: only block work on holidays when the job flag is on. */
  const effectiveNoWork = $derived(holidaysBlockWork ? noWorkSet : new Set<string>());

  /**
   * Convert the bar's *original* start + a pixel delta into a candidate calendar date.
   * Uses the viewport's column under the pointer as the calendar reference, not raw px ÷ cellW,
   * so visible weekend cells are accounted for.
   */
  function candidateDate(originalStart: string, pxDeltaFromStart: number, startCol: number): string {
    if (days.length === 0) return originalStart;
    const idxAtStart = days.findIndex(d => d.date === originalStart);
    if (idxAtStart === -1) {
      // Bar's start is outside the rendered viewport (rare — bar pre-extends). Fall back to a
      // raw cell-count step from originalStart.
      const cells = Math.round(pxDeltaFromStart / cellW);
      return addWorkdays(originalStart, cells, includeWeekends);
    }
    // Column the pointer is over now:
    const colNow = startCol + Math.round(pxDeltaFromStart / cellW);
    const clampedCol = Math.max(0, Math.min(days.length - 1, colNow));
    // Number of *viewport columns* between the bar's original start and the pointer's column.
    const colDelta = clampedCol - idxAtStart;
    // Walk the same number of viewport columns from originalStart to find the candidate date.
    const candidateIdx = idxAtStart + colDelta;
    return days[candidateIdx].date;
  }

  /** Ghost date = candidate snapped to nearest workable day for the active job. */
  function ghostDateFor(originalStart: string, pxDeltaFromStart: number, startCol: number): string {
    const candidate = candidateDate(originalStart, pxDeltaFromStart, startCol);
    return snapToNearestWorkable(candidate, effectiveNoWork, includeWeekends);
  }

  function onPointerMove(e: PointerEvent) {
    if (!store.dragState) return;
    const rawDelta = e.clientX - store.dragState.startX;
    store.dragState.liveDelta = rawDelta;
  }

  async function onPointerUp(_e: PointerEvent) {
    const d = store.dragState;
    if (!d) return;
    store.dragState = null;
    if (!store.currentJob) return;

    if (d.taskId < 0) {
      // Phase drag — shift every task in the phase by the same workday delta.
      const phaseId = -d.taskId;
      const phaseTasks = store.tasksByPhase.get(phaseId) ?? [];
      if (phaseTasks.length === 0) return;
      const first = phaseTasks[0];
      const ghost = ghostDateFor(first.start_date, d.liveDelta, d.startCol);
      if (ghost === first.start_date) return;
      // Workday delta from the first task; apply to every child preserving relative offsets.
      const workdayDelta = workdaysBetween(first.start_date, ghost, effectiveNoWork, includeWeekends);
      for (const t of phaseTasks) {
        const newStart = addWorkdays(t.start_date, workdayDelta, includeWeekends);
        await ipc.dragTask({
          job_id: store.currentJob.id,
          task_id: t.id,
          new_start_date: newStart,
        });
      }
      await ipc.touchLastSave();
      await store.openJob(store.currentJob.id);
      return;
    }

    const task = store.tasks.find(t => t.id === d.taskId);
    if (!task) return;

    if (d.zone === 'move') {
      const ghost = ghostDateFor(d.originalStart, d.liveDelta, d.startCol);
      if (ghost === d.originalStart) return;
      const result = await ipc.dragTask({
        job_id: store.currentJob.id,
        task_id: d.taskId,
        new_start_date: ghost,
      });
      store.applyDragResult(result.updated_tasks);
    } else if (d.zone === 'resize-end') {
      const ghostEnd = ghostDateFor(d.originalEnd, d.liveDelta, d.startCol);
      const newDur = Math.max(1, workdaysBetween(d.originalStart, ghostEnd, effectiveNoWork, includeWeekends) + 1);
      const updated = { ...task, duration_workdays: newDur };
      await ipc.updateTask($state.snapshot(updated));
      store.tasks = store.tasks.map(t => t.id === task.id ? updated : t);
    } else if (d.zone === 'resize-start') {
      const ghostStart = ghostDateFor(d.originalStart, d.liveDelta, d.startCol);
      const workdayDelta = workdaysBetween(d.originalStart, ghostStart, effectiveNoWork, includeWeekends);
      const newDur = Math.max(1, d.originalDuration - workdayDelta);
      const updated = { ...task, start_date: ghostStart, duration_workdays: newDur };
      await ipc.updateTask($state.snapshot(updated));
      store.tasks = store.tasks.map(t => t.id === task.id ? updated : t);
    }
    await ipc.touchLastSave();
  }

  /** Inclusive workdays between two ISO dates (positive if endIso > startIso). */
  function workdaysBetween(startIso: string, endIso: string, noWork: Set<string>, incWknd: boolean): number {
    if (startIso === endIso) return 0;
    const forward = endIso > startIso;
    let count = 0;
    let cur = startIso;
    while (cur !== endIso) {
      cur = addWorkdays(cur, forward ? 1 : -1, incWknd);
      // Skip no-work days
      while (noWork.has(cur)) {
        cur = addWorkdays(cur, forward ? 1 : -1, incWknd);
      }
      count += forward ? 1 : -1;
      if (Math.abs(count) > 10_000) break; // defensive
    }
    return count;
  }

  /** Ghost position derived from store.dragState — recomputed on every render. */
  const ghostX = $derived.by(() => {
    const d = store.dragState;
    if (!d) return -1;
    const originRef = d.zone === 'resize-end' ? d.originalEnd : d.originalStart;
    const ghost = ghostDateFor(originRef, d.liveDelta, d.startCol);
    if (ghost === originRef) return -1; // hide marker when commit delta is 0
    const x = dateToPx(ghost, cellW, days);
    return x === -1 ? -1 : x + timelineLeft;
  });

  onMount(() => {
    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    return () => {
      window.removeEventListener('pointermove', onPointerMove);
      window.removeEventListener('pointerup', onPointerUp);
    };
  });
</script>

{#if store.dragState && ghostX >= 0}
  <svg class="ghost-overlay" style="position: absolute; left: 0; top: 0; pointer-events: none; overflow: visible;">
    {#await import('./GhostMarker.svelte') then { default: GhostMarker }}
      <GhostMarker x={ghostX} top={store.dragState.rowTop} height={store.dragState.rowHeight} />
    {/await}
  </svg>
{/if}
```
